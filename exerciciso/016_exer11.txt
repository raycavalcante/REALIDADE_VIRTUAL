public class HelloWorld
{

    {   
        double a, b;
        double soma, sub, div, mod, multi;
        
        Console.Write ("Digite o primeiro numero: ");
        a = double.Parse(Console.ReadLine());
        
        Console.Write("Digite o segundo numero: ");
        b = double.Parse(Console.ReadLine());
        
        soma = a + b;
        sub = a - b;
        div = a / b;
        mod = a % b;
        multi = a * b;
